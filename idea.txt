let force = unit!(Force, Newton, 100);
let area = unit!(Area, Exponent<Meter,2>, 4);
let pressure = force / area; // Compiler erkennt: Force/Area = Pressure!
// pressure ist automatisch vom Typ Pressure (Pascal)

let pressure2 = unit!(Pressure, Prefix<Kilo,Pascal>, force / area); // pressure2 ist automatisch vom Typ Pressure (Kilopascal)
// es wird geprüft ob force / area eine Einheit ergibt, die mit Pressure kompatibel ist
// und es wird automatisch die Umwandlung von Pascal nach Kilopascal durchgeführt
